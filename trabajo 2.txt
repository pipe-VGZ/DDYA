def insertion_sort(lista, orden="asc"):
    for i in range(1, len(lista)):
        clave = lista[i]
        j = i - 1
        if orden == "asc":  
            while j >= 0 and lista[j] > clave:
                lista[j + 1] = lista[j]
                j -= 1
        else:  
            while j >= 0 and lista[j] < clave:
                lista[j + 1] = lista[j]
                j -= 1
        lista[j + 1] = clave
    return lista


n = int(input("Ingrese la cantidad de productos: "))
precios = []

for i in range(n):
    print('ingrese el precio del producto',i+1)
    precios=float(input())
    precios.append(precio)

print("Seleccione el tipo de ordenamiento:")
print(" 1 menor a mayor")
print(" 2 mayor a menor")
print('ingrese la opcion')
opcion=input()

if opcion == 1:
    precios = insertion_sort(precios, "asc")

elif opcion == 2:
    precios = insertion_sort(precios, "desc")
    
else:
    print("Opción inválida")

print(precios)
